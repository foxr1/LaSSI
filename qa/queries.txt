// q1:
// SPACE prop to haslocation for unarypred
nested match FUnaryPredicate (var(rel), var(arg), var(score), var(props))
       extend-with "extractProperties" with from : "$0" field : "SPACE" result : "loc"
       as FBinaryPredicate ("AtLocation", var(arg) , var(loc), var(score), ( "removeFromProperties" ( var(props) ) with key : "SPACE" )  )
;

// if node is of type person, do atlocation for person
nested match FUnaryPredicate (var(rel), var(arg), var(score), var(props))
       replace var(personType) with  jsonpath "$0:$.arg.type"
       extend-with "getOutgoingNodes" with variable : "personType"   result : "z"   rel : "AtLocation"
       as FBinaryPredicate ("AtLocation", var(arg) , FVariable(var(z), "ENTITY"), var(score), ( "removeFromProperties" ( var(props) ) with key : "SPACE" )  ) // added removal of space prop cuz then it will trigger the space rule to create another atlocation although maybe that isnt so bad so i dont know if i should keep it
;

// q4

// q5
// verb(src, dst) [instrument: x] -->> locatedNear(x, dst) e.g. cover(x, food)[instrument: mold] -->> LocatedNear(mold, food)
nested match  FBinaryPredicate(var(verb), var(source), var(target), var(score), var(props))
       extend-with "extractProperties" with from : "$0" field : "INSTRUMENT" result : "instr"
       as FBinaryPredicate ("LocatedNear", var(instr), var(target), var(score), ( "removeFromProperties" ( var(props) ) with key : "INSTRUMENT" )  )
       ;

// specific rule for generating HasA. if verb is use, then its a possibility that src has a dst, hence why it can use it
nested match  FBinaryPredicate("use", var(source), var(target), var(score), var(props))
       as FBinaryPredicate ("HasA", var(source), var(target), var(score), var(props)  )
       ;

// this is a rule that doesnt always generate a predicate that makes sense
// absorb(blotter, ink) -> atlocation(ink, blotter) makes sense here and it will be already in the ontology
// isa(apple, fruit) -> atlocation(fruit, apple) doesn't make sense and it wont be an existing relation in the ontology
nested match  FBinaryPredicate(var(verb), var(source), var(target), var(score), var(props))
       as FBinaryPredicate ("AtLocation", var(target), var(source), var(score), var(props))
       ;
